#define main hostsim_main
#include HOSTSIM_PATH
#undef main
static void fresh(void)
{
    memset(trk,0,sizeof trk);memset(&song,0,sizeof song);host_tracks_init();
    memset(heavy_pool,0,sizeof heavy_pool);memset(heavy_owner,0,sizeof heavy_owner);
#if !TEST_SHARED_SLICE
    memset(slc_rbuf,0,sizeof slc_rbuf);memset(slc_owner,0,sizeof slc_owner);
#endif
}
int main(void)
{
    for(uint32_t source=0;source<=4;source+=4)for(uint32_t rev=0;rev<2;rev++)for(uint32_t mode=V_POLY;mode<=V_UNISON;mode++){
        fresh();uint32_t audible=0;
        for(uint32_t p=0;p<NPART;p++){
            host_preset(&trk[p],13,0);trk[p].p[P_VOICE]=mode;
            trk[p].p[P_E0]=source;trk[p].p[P_E4]=SLC_LOOP;trk[p].p[P_E5]=rev;
            if(mode!=V_UNISON||p==0)trk_note_on(&trk[p],SLC_BASE+p,100);
        }
        for(uint32_t b=0;b<256;b++){
            if(b==100||b==200)for(uint32_t p=0;p<NPART;p++)if(mode!=V_UNISON||p==0)trk_note_on(&trk[p],SLC_BASE+p,100);
            for(uint32_t p=0;p<NPART;p++)engine_block(&trk[p]);
            for(uint32_t p=0;p<NPART;p++){
                int32_t out[CTL];track_render(&trk[p],out,CTL);fwrite(out,sizeof out,1,stdout);
                for(uint32_t i=0;i<CTL;i++)audible+=out[i]!=0;
            }
        }
        fprintf(stderr,"source %u reverse %u mode %u audible samples %u\n",source,rev,mode,audible);
        if(!audible)return 1;
    }
    return 0;
}
