def hdr(obj,seq,payload):
 h=struct.pack('<4sIIIIII',b'TEST',obj,seq,len(payload),zlib.crc32(payload),0,0)
 return h+struct.pack('<I',zlib.crc32(h))
def scan(blocks):
 winners={}
 for i,b in enumerate(blocks):
  h=b[:32]
  if h[:4]!=b'TEST' or zlib.crc32(h[:28])!=struct.unpack_from('<I',h,28)[0]:continue
  _,obj,seq,n,crc,a,c=struct.unpack('<4sIIIIII',h[:28])
  if obj>=OBJECTS or n>7936 or a or c:continue
  payload=bytes(b[OFFSET:OFFSET+n])
  if zlib.crc32(payload)!=crc:continue
  if obj not in winners or 0<(seq-winners[obj][1])&0xffffffff<0x80000000:winners[obj]=(i,seq,payload)
 return winners
