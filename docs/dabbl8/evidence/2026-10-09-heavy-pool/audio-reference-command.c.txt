#define main hostsim_main
#include HOSTSIM_PATH
#undef main
static void reset(void)
{
    memset(trk,0,sizeof trk);memset(&song,0,sizeof song);host_tracks_init();
#if TEST_SHARED_HEAVY
    memset(heavy_owner,0,sizeof heavy_owner);memset(heavy_pool,0,sizeof heavy_pool);
#else
    memset(gr_p,0,sizeof gr_p);memset(phys_owner,0,sizeof phys_owner);memset(phys_slot,0,sizeof phys_slot);memset(drum_kit,0,sizeof drum_kit);
#endif
}
int main(void)
{
    for(uint32_t e=8;e<=10;e++){
        reset();uint32_t h=2166136261u,nonzero=0;
        for(uint32_t p=0;p<NPART;p++){
            host_preset(&trk[p],e,0);trk[p].p[P_VOICE]=V_POLY;
            trk_note_on(&trk[p],e==10?36:60+p,100);
        }
        for(uint32_t b=0;b<1000;b++){
            if(b==200||b==500)for(uint32_t p=0;p<NPART;p++)trk_note_on(&trk[p],e==10?36:60+p,100);
            for(uint32_t p=0;p<NPART;p++)engine_block(&trk[p]);
            for(uint32_t p=0;p<NPART;p++){
                int32_t out[CTL];track_render(&trk[p],out,CTL);fwrite(out,sizeof out,1,stdout);
                for(uint32_t i=0;i<CTL;i++){h=(h^(uint32_t)out[i])*16777619u;nonzero+=out[i]!=0;}
            }
        }
        fprintf(stderr,"engine %u audio %08x nonzero %u\n",e,h,nonzero);
    }
    return 0;
}
